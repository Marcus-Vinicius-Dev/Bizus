def testar_arquivo(caminho_teste_arquivo, sep=None, encoding='utf-8-sig'):     # aceita CSV/XLSX/PARQUET
    # 1. Existe?
    if not caminho_teste_arquivo or not os.path.exists(caminho_teste_arquivo): # valida caminho e existência
        print(Fore.RED + f"❌ Arquivo não encontrado: {caminho_teste_arquivo}") # aviso vermelho
        return None                                                            # sai
    extensao = os.path.splitext(caminho_teste_arquivo)[1].lower()              # pega extensão minúscula
    # 2. Tenta ler conforme extensão
    try:                                                                       # bloco protegido
        if extensao == '.csv':                                                 # ramo CSV
            df = pd.read_csv(caminho_teste_arquivo, sep=sep, engine='python', encoding=encoding)  # lê CSV
        elif extensao == '.xlsx':                                              # ramo XLSX
            df = pd.read_excel(caminho_teste_arquivo, engine='openpyxl')       # lê XLSX
        elif extensao == '.parquet':                                           # ramo PARQUET
            df = pd.read_parquet(caminho_teste_arquivo, engine='pyarrow')      # lê PARQUET
        else:                                                                  # extensão não suportada
            print(Fore.RED + f"❌ Extensão não suportada: {extensao}")         # aviso vermelho
            return None                                                        # sai
    except UnicodeDecodeError as e:                                            # falha de encoding (só CSV)
        print(Fore.YELLOW + f"⚠️  {encoding} falhou ({e}), tentando latin-1...")  # avisa fallback
        df = pd.read_csv(caminho_teste_arquivo, sep=sep, engine='python', encoding='latin-1')  # fallback
    except pd.errors.ParserError as e:                                         # falha de parser (só CSV)
        print(Fore.RED + f"❌ Erro no parser: {e}")                             # aviso vermelho
        return None                                                            # sai
    except Exception as e:                                                     # qualquer outro erro (XLSX/Parquet)
        print(Fore.RED + f"❌ Erro inesperado: {e}")                            # aviso vermelho
        return None                                                            # sai
    # 3. Mostra info
    print(Fore.GREEN + f"\n📊 ARQUIVO VALIDADO:")                              # título verde
    print(Fore.WHITE + f"   Arquivo:  {os.path.basename(caminho_teste_arquivo)}")  # nome do arquivo
    print(Fore.WHITE + f"   Linhas:   {len(df):,}")                            # total de linhas
    print(Fore.WHITE + f"   Colunas:  {len(df.columns)}")                      # total de colunas
    print(Fore.WHITE + f"   Nulos:    {df.isna().sum().sum():,}")              # total de nulos
    # 4. Nulos por coluna
    print(Fore.GREEN + f"\n📊 NULOS POR COLUNA:")                              # título verde
    for i, col in enumerate(df.columns, 1):                                    # percorre colunas
        nulos = df[col].isna().sum()                                           # nulos da coluna
        pct = (nulos / len(df) * 100) if len(df) > 0 else 0                    # % de nulos
        cor = Fore.RED if pct > 50 else Fore.YELLOW if pct > 10 else Fore.CYAN # cor dinâmica
        print(cor + f"   {i:3d}. {col:30s} | Nulos: {nulos:6,} ({pct:5.1f}%)") # linha formatada
    return df                                                                  # retorna DataFrame

def conv_para_parquet(caminho_entrada, pasta_saida, sep=None, encoding='utf-8-sig'):  # aceita CSV e XLSX
    # 0. Valida entrada
    if not caminho_entrada or not os.path.isfile(caminho_entrada):                    # valida existência
        print(Fore.RED + Style.BRIGHT + f"❌ arquivo não encontrado: '{caminho_entrada}'")  # aviso
        return None                                                                   # sai
    if not pasta_saida or not os.path.isdir(pasta_saida):                             # valida pasta de saída
        print(Fore.RED + Style.BRIGHT + f"❌ pasta de saída não existe: '{pasta_saida}'")   # aviso
        return None                                                                   # sai

    extensao = os.path.splitext(caminho_entrada)[1].lower()                           # extensão minúscula
    nome_base = os.path.splitext(os.path.basename(caminho_entrada))[0]                # nome sem extensão
    nome_parquet = nome_base + '.parquet'                                             # nome do parquet
    caminho_saida = os.path.join(pasta_saida, nome_parquet)                           # caminho final

    print(Fore.YELLOW + f"\n⚙️  CONVERTENDO → PARQUET (streaming pyarrow)...")        # título
    print(Fore.WHITE + f"   Arquivo:  {os.path.basename(caminho_entrada)}")           # nome entrada
    print(Fore.WHITE + f"   Tamanho:  {os.path.getsize(caminho_entrada) / 1024**3:.2f} GB")  # tamanho GB
    print(Fore.WHITE + f"   Extensão: {extensao}")                                    # extensão detectada

    # 1. Se for XLSX → converte para CSV temporário primeiro
    if extensao == '.xlsx':                                                           # ramo XLSX
        print(Fore.YELLOW + f"\n📊 XLSX detectado → convertendo para CSV temporário...")  # aviso
        caminho_csv_temp = os.path.join(pasta_saida, nome_base + '_TEMP.csv')         # caminho do CSV temp
        try:                                                                          # bloco protegido
            df = pd.read_excel(caminho_entrada, dtype=str, engine='openpyxl')         # lê XLSX como string
            df.to_csv(caminho_csv_temp, sep=';', index=False, encoding='utf-8-sig')   # grava CSV temporário
            caminho_entrada = caminho_csv_temp                                        # aponta para o CSV temp
            extensao = '.csv'                                                         # passa a tratar como CSV
            print(Fore.GREEN + f"   ✅ CSV temporário criado: {os.path.basename(caminho_csv_temp)}")  # confirma
        except Exception as e:                                                        # qualquer erro
            print(Fore.RED + f"❌ Erro ao converter XLSX → CSV: {e}")                  # aviso
            return None                                                               # sai
    elif extensao != '.csv':                                                          # não suportado
        print(Fore.RED + Style.BRIGHT + f"❌ extensão não suportada: '{extensao}'")    # aviso
        return None                                                                   # sai

    # 2. Descobre esquema do CSV (amostra de 1000 linhas)
    print(Fore.YELLOW + f"\n📊 ANALISANDO TIPOS DAS COLUNAS...")                     # título
    df_amostra = pd.read_csv(                                                        # lê amostra
        caminho_entrada,                                                             # caminho CSV
        sep=sep,                                                                     # separador (None = auto)
        engine='python',                                                             # necessário p/ sep=None
        encoding=encoding,                                                           # encoding
        nrows=1000,                                                                  # só 1000 linhas
        dtype=str                                                                    # força string
    )
    column_types = {col: pa.string() for col in df_amostra.columns}                  # tudo string (seguro)
    # detecta o separador efetivo via csv.Sniffer (funciona com sep=None)
    if sep is None:                                                                  # se não foi passado
        with open(caminho_entrada, 'r', encoding=encoding, errors='ignore') as f:    # abre amostra
            primeira_linha = f.readline()                                            # lê 1ª linha
        sep_efetivo = csv.Sniffer().sniff(primeira_linha, delimiters=';,\t|').delimiter  # detecta
    else:                                                                            # se foi passado
        sep_efetivo = sep                                                            # usa o informado
    print(Fore.WHITE + f"   Separador detectado: '{sep_efetivo}'")                    # informa separador
    print(Fore.WHITE + f"   Colunas analisadas: {len(column_types)}")                 # total colunas
    print(Fore.WHITE + f"   Colunas como string: {len(column_types)}")                # todas string

    # 3. Opções de leitura pyarrow
    read_options = pv.ReadOptions(block_size=100 * 1024 * 1024)                       # 100 MB por bloco
    parse_options = pv.ParseOptions(delimiter=sep_efetivo)                            # delimitador efetivo
    convert_options = pv.ConvertOptions(                                              # opções de conversão
        column_types=column_types,                                                    # força tipos
        strings_can_be_null=True                                                      # "" vira nulo
    )

    # 4. Streaming CSV → Parquet
    print(Fore.YELLOW + f"\n⚙️  CONVERTENDO...")                                      # título
    with pv.open_csv(                                                                 # abre CSV em streaming
        caminho_entrada,                                                              # CSV de entrada
        read_options=read_options,                                                    # opções de leitura
        parse_options=parse_options,                                                  # opções de parsing
        convert_options=convert_options                                               # opções de conversão
    ) as reader:                                                                      # reader pyarrow
        writer = None                                                                 # writer lazy
        total_linhas = 0                                                              # contador
        with tqdm(desc="Convertendo", unit=" linhas") as pbar:                        # barra de progresso
            for batch in reader:                                                      # percorre batches
                table = pa.Table.from_batches([batch])                                # batch → table
                if writer is None:                                                    # primeira vez
                    writer = pq.ParquetWriter(caminho_saida, table.schema)            # cria writer
                writer.write_table(table)                                             # grava table
                total_linhas += len(table)                                            # soma linhas
                pbar.update(len(table))                                               # atualiza barra
        if writer:                                                                    # se writer criado
            writer.close()                                                            # fecha writer

    # 5. Remove CSV temporário (se veio de XLSX)
    if caminho_entrada.endswith('_TEMP.csv') and os.path.isfile(caminho_entrada):
        removido = False
        for tentativa in range(3):  # ← Tenta 3 vezes
            try:
                time.sleep(0.5)  # ← Aguarda entre tentativas
                os.remove(caminho_entrada)
                print(Fore.YELLOW + f"   🧹 CSV temporário removido")
                removido = True
                break
            except PermissionError:
                if tentativa < 2:
                    print(Fore.YELLOW + f"   ⏳ Tentativa {tentativa+1}/3...")
                else:
                    print(Fore.YELLOW + f"   ⚠️ Não foi possível remover temp")
                    print(Fore.WHITE + f"   💡 Remova manualmente: {caminho_entrada}")

# CHAMAR conv_para_parquet

def resolver_caminho_origem(pasta_origem, arquivo_origem):                     # helper reutilizável

    arquivo_selecionado = dicionario_arquivos_origem.get(pasta_origem, {}).get(arquivo_origem, None)      # resolve arquivo
    if pasta_base and arquivo_selecionado:                                                                # ambos existem
        return os.path.join(pasta_base, arquivo_selecionado)                                              # caminho montado
    return None                                                                                        # senão None

def _detectar_linha_cabecalho(df_bruto, max_linhas=10):                 # helper existente
    for i in range(min(max_linhas, len(df_bruto))):                     # percorre as primeiras linhas
        linha = df_bruto.iloc[i]                                        # linha atual
        nao_nulos = linha.notna().sum()                                 # conta não-nulos
        if nao_nulos >= len(df_bruto.columns) * 0.5:                    # maioria preenchida
            return i                                                    # achou cabeçalho
    return 0                                                            # fallback: 1ª linha

def ler_e_normalizar(caminho_arquivo, detectar=True):                   # ← NOVO: normalizador
    ext = os.path.splitext(caminho_arquivo)[1].lower()                  # extensão minúscula
    if ext == '.csv':                                                   # ramo CSV
        df_bruto = pd.read_csv(caminho_arquivo, sep=None,
                               engine='python', dtype=str,
                               encoding='utf-8-sig')                    # CSV cru
    elif ext == '.xlsx':                                                # ramo XLSX
        df_bruto = pd.read_excel(caminho_arquivo, dtype=str,
                                 engine='openpyxl')                     # XLSX cru
    elif ext == '.parquet':                                             # ramo PARQUET
        df_bruto = pd.read_parquet(caminho_arquivo,
                                   engine='pyarrow')                    # Parquet cru
    else:                                                               # não suportado
        print(Fore.RED + Style.BRIGHT + f"extensão não suportada: '{ext}'")  # aviso
        return None                                                     # sai
    if detectar:                                                        # seletor de cabeçalho
        idx_cab = _detectar_linha_cabecalho(df_bruto)                   # detecta linha do cabeçalho
        df = df_bruto.iloc[idx_cab + 1:].reset_index(drop=True)         # dados abaixo do cabeçalho
        df.columns = [str(c).replace('\n', ' ').strip() if pd.notna(c)
                      else f'COLUNA_{i}'
                      for i, c in enumerate(df_bruto.iloc[idx_cab])]    # nomes limpos
        return df                                                       # DataFrame normalizado
    return df_bruto                                                     # senão, DataFrame cru

    if df is None:                                                      # se falhou

    df = df.head(n)                                                     # corta TOP N
    print(Fore.GREEN + Style.BRIGHT + f"TOP {n} de '{os.path.basename(caminho_arquivo)}' ({len(df)} linhas lidas)")  # cabeçalho
    print(Fore.CYAN + df.to_string(index=False))                        # imprime sem índice
    return df                                                           # retorna DataFrame
