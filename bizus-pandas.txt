def testar_csv(caminho_teste_csv, sep=';', encoding='utf-8-sig'):
    """Valida um CSV rapidamente"""
    
    # 1. Existe?
    if not os.path.exists(caminho_teste_csv):
        print(Fore.RED + f"❌ Arquivo não encontrado: {caminho_teste_csv}")
        return None
    
    # 2. Tenta ler
    try:
        df = pd.read_csv(caminho_teste_csv, sep=sep, encoding=encoding)
    except pd.errors.ParserError as e:
        print(Fore.RED + f"❌ Erro no parser: {e}")
        return None
    except UnicodeDecodeError:
        df = pd.read_csv(caminho_teste_csv, sep=sep, encoding='latin-1')
    
    # 3. Mostra info
    print(Fore.GREEN + f"\n📊 ARQUIVO VALIDADO:")
    print(Fore.WHITE + f"   Arquivo:  {os.path.basename(caminho_teste_csv)}")
    print(Fore.WHITE + f"   Linhas:   {len(df):,}")
    print(Fore.WHITE + f"   Colunas:  {len(df.columns)}")
    print(Fore.WHITE + f"   Nulos:    {df.isna().sum().sum():,}")
    
        # 4. Nulos por coluna
    print(Fore.GREEN + f"\n📊 NULOS POR COLUNA:")
    for i, col in enumerate(df.columns, 1):
        nulos = df[col].isna().sum()
        pct = (nulos / len(df) * 100) if len(df) > 0 else 0
        cor = Fore.RED if pct > 50 else Fore.YELLOW if pct > 10 else Fore.CYAN
        print(cor + f"   {i:3d}. {col:30s} | Nulos: {nulos:6,} ({pct:5.1f}%)")

    return df

def conv_csv_to_parquet(caminho_csv, nome_parquet=None, sep=';', encoding='utf-8-sig'):
    if nome_parquet is None:
        nome_parquet = os.path.splitext(os.path.basename(caminho_csv))[0] + '.parquet'
    
    caminho_saida = os.path.join(pasta_parquet, nome_parquet)
    df = pd.read_csv(caminho_csv, sep=sep, encoding=encoding)
    df.to_parquet(caminho_saida, index=False, engine='pyarrow')
    
    print(Fore.GREEN + f"\n✅ CONVERTIDO:")
    print(Fore.WHITE + f"   CSV:     {os.path.basename(caminho_csv)}")
    print(Fore.WHITE + f"   Parquet: {nome_parquet}")
    print(Fore.WHITE + f"   Linhas:  {len(df):,}")
    print(Fore.WHITE + f"   Colunas: {len(df.columns)}")
    
    return caminho_saida

# ============================================================================
# VERIFICAR PARQUET (máximo de informações)
# ============================================================================

def verificar_parquet(caminho_parquet):
    """Extrai o máximo de informações de um arquivo Parquet"""
    
    # 1. Verifica se existe
    if not os.path.exists(caminho_parquet):
        print(Fore.RED + f"❌ Parquet não encontrado: {caminho_parquet}")
        return None
    
    print(Fore.GREEN + "\n" + "="*70)
    print(Fore.GREEN + "📊 VERIFICAÇÃO COMPLETA DO PARQUET")
    print(Fore.GREEN + "="*70)
    
    # =========================================================================
    # 1. INFORMAÇÕES DO ARQUIVO
    # =========================================================================
    tamanho_bytes = os.path.getsize(caminho_parquet)
    tamanho_kb = tamanho_bytes / 1024
    tamanho_mb = tamanho_kb / 1024
    
    print(Fore.YELLOW + "\n📁 ARQUIVO:")
    print(Fore.WHITE + f"   Nome:     {os.path.basename(caminho_parquet)}")
    print(Fore.WHITE + f"   Pasta:    {os.path.dirname(caminho_parquet)}")
    print(Fore.WHITE + f"   Tamanho:  {tamanho_kb:,.2f} KB ({tamanho_mb:.4f} MB)")
    
    # =========================================================================
    # 2. METADADOS (sem carregar dados)
    # =========================================================================
    import pyarrow.parquet as pq
    parquet_file = pq.ParquetFile(caminho_parquet)
    
    print(Fore.YELLOW + "\n📋 METADADOS:")
    print(Fore.WHITE + f"   Linhas:     {parquet_file.metadata.num_rows:,}")
    print(Fore.WHITE + f"   Colunas:    {parquet_file.metadata.num_columns}")
    print(Fore.WHITE + f"   Blocos internos de armazenamento: {parquet_file.metadata.num_row_groups}")
    print(Fore.WHITE + f"   Criado por: {parquet_file.metadata.created_by}")
    
    # =========================================================================
    # 3. CARREGAR O DATAFRAME
    # =========================================================================
    print(Fore.YELLOW + "\n📖 CARREGANDO DATAFRAME...")
    df = pd.read_parquet(caminho_parquet)
    
    # =========================================================================
    # 4. ESTRUTURA
    # =========================================================================
    memoria_bytes = df.memory_usage(deep=True).sum()
    memoria_mb = memoria_bytes / 1024 / 1024
    
    print(Fore.YELLOW + "\n📊 ESTRUTURA:")
    print(Fore.WHITE + f"   Linhas:           {len(df):,}")
    print(Fore.WHITE + f"   Colunas:          {len(df.columns)}")
    print(Fore.WHITE + f"   Células totais:   {df.size:,}")
    print(Fore.WHITE + f"   Memória (RAM):    {memoria_mb:.4f} MB")
    
    # =========================================================================
    # 5. NULOS
    # =========================================================================
    total_nulos = df.isna().sum().sum()
    pct_nulos = (total_nulos / df.size * 100) if df.size > 0 else 0
    
    print(Fore.YELLOW + "\n📊 NULOS:")
    print(Fore.WHITE + f"   Total de nulos:   {total_nulos:,}")
    print(Fore.WHITE + f"   Percentual:       {pct_nulos:.2f}%")
    print(Fore.WHITE + f"   Colunas c/ nulos: {(df.isna().sum() > 0).sum()} de {len(df.columns)}")
    
    # =========================================================================
    # 6. CABEÇALHO COM DETALHES
    # =========================================================================
    print(Fore.YELLOW + "\n📋 CABEÇALHO DETALHADO:")
    print(Fore.WHITE + f"   {'#':<4} {'Coluna':<30} {'Tipo':<12} {'Nulos':<10} {'%':<8} {'Únicos':<8}")
    print(Fore.WHITE + "   " + "-"*80)
    
    for i, col in enumerate(df.columns, 1):
        dtype = str(df[col].dtype)
        nulos = df[col].isna().sum()
        pct = (nulos / len(df) * 100) if len(df) > 0 else 0
        unicos = df[col].nunique()
        
        # Cor baseada em nulos
        if pct > 50:
            cor = Fore.RED
        elif pct > 10:
            cor = Fore.YELLOW
        else:
            cor = Fore.CYAN
        
        print(cor + f"   {i:<4} {col:<30} {dtype:<12} {nulos:<10,} {pct:>6.1f}% {unicos:<8,}")
    
    # =========================================================================
    # 7. TIPOS DE DADOS (resumo)
    # =========================================================================
    print(Fore.YELLOW + "\n📊 TIPOS DE DADOS:")
    tipos = df.dtypes.value_counts()
    for tipo, qtd in tipos.items():
        print(Fore.CYAN + f"   {str(tipo):<15} → {qtd} coluna(s)")
    
    # =========================================================================
    # 8. DUPLICATAS
    # =========================================================================
    duplicatas_linhas = df.duplicated().sum()
    duplicatas_colunas = df.columns[df.columns.duplicated()].tolist()
    
    print(Fore.YELLOW + "\n📊 DUPLICATAS:")
    print(Fore.WHITE + f"   Linhas duplicadas:  {duplicatas_linhas:,}")
    if duplicatas_colunas:
        print(Fore.RED + f"   Colunas duplicadas: {duplicatas_colunas}")
    else:
        print(Fore.GREEN + f"   Colunas duplicadas: Nenhuma")
    
    # =========================================================================
    # 9. COLUNAS COM PROBLEMAS
    # =========================================================================
    print(Fore.YELLOW + "\n⚠️  COLUNAS COM PROBLEMAS:")
    
    problemas = []
    
    # 9.1 Colunas 100% vazias
    vazias = [col for col in df.columns if df[col].isna().all()]
    if vazias:
        problemas.append(f"   100% vazias: {vazias}")
    
    # 9.2 Colunas com valor único (constantes)
    constantes = [col for col in df.columns if df[col].nunique() == 1]
    if constantes:
        problemas.append(f"   Valor único: {constantes}")
    
    # 9.3 Colunas com muitos nulos (>50%)
    muitos_nulos = [col for col in df.columns if (df[col].isna().sum() / len(df) * 100) > 50]
    if muitos_nulos:
        problemas.append(f"   +50% nulos: {muitos_nulos}")
    
    if problemas:
        for p in problemas:
            print(Fore.YELLOW + p)
    else:
        print(Fore.GREEN + "   ✅ Nenhum problema detectado")
    
    # =========================================================================
    # 10. PRIMEIRAS LINHAS
    # =========================================================================
    print(Fore.YELLOW + "\n📊 PRIMEIRAS 5 LINHAS:")
    print(Fore.CYAN + f"\n{df.head()}")
    
    # =========================================================================
    # 11. RESUMO FINAL
    # =========================================================================
    print(Fore.GREEN + "\n" + "="*70)
    print(Fore.GREEN + "✅ VERIFICAÇÃO CONCLUÍDA")
    print(Fore.GREEN + "="*70)
    
    return df
